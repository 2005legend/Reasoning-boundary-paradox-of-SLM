# STEP 4: PREPARE DATASET WITH CLUSTER IDS

print("⏳ STEP 4/8: Preparing training dataset with cluster IDs...")

# Get problems and answers
train_problems = gsm8k_train['problems']
train_answers = gsm8k_train['answers']

# For smoke tier, use subset

    train_problems = train_problems[:500]

# Format prompts with XML template
formatted_prompts = batch_format_prompts(train_problems, include_example=True)

# Create dataset WITH CLUSTER IDS
print(f"   Assigning cluster IDs to {len(formatted_prompts)} prompts...")

train_data = []
cluster_id_count = 0
missing_cluster_count = 0

for prompt, problem, answer in zip(formatted_prompts, train_problems, train_answers):
    # Get cluster ID from assignments
    cluster_id = cluster_assignments.get(problem, 0)  # Default to 0 if not found
    
    if problem not in cluster_assignments:
        missing_cluster_count += 1
    
    train_data.append({
        "prompt": prompt,
        "problem": problem,
        "ground_truth": answer,
        "cluster_id": cluster_id  # CRITICAL: Add cluster ID for CB-GRPO
    })
    cluster_id_count += 1

print(f"✅ Dataset prepared: {len(train_data)} examples")
print(f"   Cluster IDs assigned: {cluster_id_count}")
if missing_cluster_count > 0:
    print(f"   ⚠️  Missing cluster assignments: {missing_cluster_count} (defaulted to cluster 0)")

# Create HuggingFace Dataset
train_dataset = Dataset.from_list(train_data)

# Verify cluster_id field exists
if 'cluster_id' not in train_dataset.column_names:
    print("❌ ERROR: 'cluster_id' field missing from dataset!")
    raise ValueError("Dataset must have 'cluster_id' field for CB-GRPO")

print(f"   Dataset columns: {train_dataset.column_names}")
