def unbiased_pass_at_k(n, c, k):
    """Compute unbiased Pass@k estimator."""
    if n - c < k:
        return 1.0
    return 1.0 - comb(n - c, k) / comb(n, k)

def load_model_forced_gpu(checkpoint_path=None):
    """Load model on GPU with 4-bit quantization."""
    bnb_config = BitsAndBytesConfig(
        load_in_4bit=True,
        bnb_4bit_quant_type="nf4",
        bnb_4bit_compute_dtype=torch.float16,
        bnb_4bit_use_double_quant=True
    )
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_NAME,
        quantization_config=bnb_config,
        device_map={"": 0},  # forces cuda:0
        trust_remote_code=True,
    )
    tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME, trust_remote_code=True)
    if tokenizer.pad_token is None:
        tokenizer.pad_token = tokenizer.eos_token
    
    if checkpoint_path is not None:
        print(f"  Loading LoRA adapters from: {checkpoint_path}")
        model = PeftModel.from_pretrained(model, checkpoint_path)
    else:
        print(f"  Using base model (no training)")
    
    model.eval()
    total_params = sum(p.numel() for p in model.parameters())
    trainable_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    print(f"✅ Model loaded on {next(model.parameters()).device}")
    print(f"   Total params: {total_params:,} | Trainable: {trainable_params:,}")
    return model, tokenizer

def format_prompt(problem):
    return f"""Solve this math problem step by step.

Problem: {problem}

Provide your answer in this format:
<reasoning>
[Your reasoning]
</reasoning>
<answer>
[Final number]
</answer>

Solution:"""

def extract_answer(completion):
    m = re.search(r'<answer>\s*(-?\d+(?:[,\.\d]*)?)\s*</answer>', completion, re.IGNORECASE)
    return m.group(1).replace(',', '') if m else None

def check_correct(extracted, ground_truth):
    if extracted is None:
        return False
    try:
        return abs(float(extracted) - float(str(ground_truth).replace(',', ''))) < 1e-4
    except ValueError:
        return False

@torch.no_grad()
def generate_n_samples(model, tokenizer, prompt, n, max_new_tokens=400, temperature=1.0):
    """Generate n samples in one batched call (much faster than looping)."""
    inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
    outputs = model.generate(
        **inputs,
        max_new_tokens=max_new_tokens,
        do_sample=True,
        temperature=temperature,
        top_p=0.95,
        num_return_sequences=n,
        pad_token_id=tokenizer.pad_token_id,
    )
    input_len = inputs['input_ids'].shape[1]
    return [tokenizer.decode(o[input_len:], skip_special_tokens=True) for o in outputs]

def parse_gsm8k_answer(answer_text):
    return answer_text.split('####')[-1].strip().replace(',', '') if '####' in answer_text else ""

def run_eval(model, tokenizer, problems, ground_truths, dataset_name, exp_name):
    """Run Pass@k evaluation on a dataset."""
    results = {
        "experiment": exp_name,
        "dataset": dataset_name,
        "n_samples": N_SAMPLES,
        "per_problem": [],
        "timestamp": str(datetime.now())
    }
    all_pass_at_k = {k: [] for k in K_VALUES}

    for i, (prob, gt) in enumerate(zip(problems, ground_truths)):
        completions = generate_n_samples(model, tokenizer, format_prompt(prob), N_SAMPLES)
        correct_flags = [check_correct(extract_answer(c), gt) for c in completions]
        c = sum(correct_flags)
        problem_pass_at_k = {k: unbiased_pass_at_k(N_SAMPLES, c, k) for k in K_VALUES}
        for k in K_VALUES:
            all_pass_at_k[k].append(problem_pass_at_k[k])

        results["per_problem"].append({
            "problem": prob,
            "ground_truth": gt,
            "n_correct": c,
            "pass_at_k": problem_pass_at_k,
        })
        if (i + 1) % 5 == 0:
            running = {k: sum(v)/len(v) for k, v in all_pass_at_k.items()}
            print(f"  [{i+1}/{len(problems)}] running Pass@k: "
                  f"{', '.join(f'{k}={v:.2f}' for k,v in running.items())}")

    results["aggregate_pass_at_k"] = {k: sum(v)/len(v) for k, v in all_pass_at_k.items()}
    return results
