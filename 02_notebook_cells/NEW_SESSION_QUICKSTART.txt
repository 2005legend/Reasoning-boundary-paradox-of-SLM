```python
missing = []

    module_path = modules_dir / module
    if module_path.exists():
        print(f"✅ Found: {module}")
    else:
        print(f"❌ Missing: {module}")
        missing.append(module)

if missing:
    print("\nACTION REQUIRED:")
    print("Upload these files to your Drive:")
    for m in missing:
        print(f"  - {m}")
    print(f"\nDestination: {modules_dir}")
    raise FileNotFoundError(f"Missing modules: {missing}")

# Import modules

    from cbgrpo_config import CBGRPOConfig, create_cbgrpo_config_from_experiment
    from cbgrpo_trainer import CBGRPOTrainer
    print("\n✅ CB-GRPO modules imported successfully!")
```

```python
# ============================================================================
# EXPERIMENT CONFIGURATION
# ============================================================================

from dataclasses import dataclass
from typing import List, Optional

@dataclass
class ExperimentConfig:
    """Configuration for RLVR experiments."""
    exp_name: str
    model_size: str
    gate_type: str
    training_steps: int
    batch_size: int = 2
    gradient_accumulation_steps: int = 4
    learning_rate: float = 5e-6
    lr_scheduler_type: str = "cosine"
    warmup_ratio: float = 0.1
    checkpoint_interval: int = 100
    eval_interval: int = 100
    log_interval: int = 10
    rollouts_per_prompt: int = 4
    max_new_tokens: int = 512
    temperature: float = 0.7
    top_p: float = 0.9
    lora_r: int = 16
    lora_alpha: int = 32
    lora_target_modules: List[str] = None
    random_seed: int = 42
    
    def __post_init__(self):
        if self.lora_target_modules is None:
            self.lora_target_modules = ["q_proj", "v_proj", "k_proj", "o_proj"]
    
    def to_dict(self):
        return self.__dict__

# Set experiment configuration
EXPERIMENT_NAME = "exp2_n1_cbgrpo_0.5B"
USE_SMOKE_TIER = True  # Set to False for full training (800 steps)

# Create config
if USE_SMOKE_TIER:
    config = ExperimentConfig(
        exp_name=f"{EXPERIMENT_NAME}_smoke",
        model_size="0.5B",
        gate_type="vanilla",
        training_steps=100,
        checkpoint_interval=50,
        batch_size=2,
        gradient_accumulation_steps=4
    )
else:
    config = ExperimentConfig(
        exp_name=EXPERIMENT_NAME,
        model_size="0.5B",
        gate_type="vanilla",
        training_steps=800,
        checkpoint_interval=100
    )

print("=" * 60)
print("EXPERIMENT CONFIGURATION")
print("=" * 60)
print(f"Experiment: {config.exp_name}")
print(f"Model: Qwen2.5-{config.model_size}-Instruct")
print(f"Training steps: {config.training_steps}")
print(f"Smoke tier: {USE_SMOKE_TIER}")
print("=" * 60)
```
