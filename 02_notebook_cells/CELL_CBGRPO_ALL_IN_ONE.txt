print("⏳ STEP 6/9: Defining reward function...")

def compute_reward_for_grpo(prompts, completions, ground_truth=None, problem=None, **kwargs):
    rewards = []
    for i, completion in enumerate(completions):
        reward = 0.0
        if '<reasoning>' in completion and '</reasoning>' in completion:
            reward += 0.1
        if '<answer>' in completion and '</answer>' in completion:
            reward += 0.1
        if ground_truth is not None:
            m = re.search(r'<answer>\s*(-?\d+(?:[,\.\d]*)?)\s*</answer>', completion, re.IGNORECASE)
            if m:
                extracted = m.group(1).replace(',', '')
                try:
                    if abs(float(extracted) - float(str(ground_truth[i]).replace(',', ''))) < 1e-4:
                        reward += 0.8
                except ValueError:
                    pass
        # Outlier detection for debugging recurring anomalous loss values
        if reward > 0.9 or reward < 0:
            print(f"[REWARD OUTLIER] reward={reward:.4f} completion={completion[:150]!r}")
        rewards.append(reward)
    return rewards
print("✅ Reward function defined\n")

# ==============================================================================
# STEP 7: CB-GRPO — zero-init spend, min-visits warm-up gate, live lazy sampler
# ==============================================================================
print("⏳ STEP 7/9: Setting up CB-GRPO spend tracker and lazy sampler...")

shared_spend_ema = np.zeros(N_CLUSTERS)          # <-- FIX 1: was np.ones(...)
shared_visit_counts = np.zeros(N_CLUSTERS, dtype=int)   # <-- FIX 2: warm-up tracking
sampled_cluster_log = deque(maxlen=200)

class CBGRPOSpendTracker:
    """Updates spend_ema and visit_counts live from rewards. Does NOT scale
    rewards themselves — GRPO's group-relative normalization would cancel out
    a uniform per-group scale, so throttling happens via sampling frequency."""
    def __init__(self, base_reward_fn, cluster_map, spend_ema, visit_counts, ema_alpha=0.01):
        self.base_reward_fn = base_reward_fn
        self.cluster_map = cluster_map
        self.spend_ema = spend_ema
        self.visit_counts = visit_counts
        self.ema_alpha = ema_alpha
        self.step = 0

    def __call__(self, prompts, completions, problem=None, ground_truth=None, **kwargs):
        rewards = self.base_reward_fn(prompts, completions, ground_truth=ground_truth, **kwargs)
        problems = problem if problem is not None else prompts
        for r, p in zip(rewards, problems):
            cid = self.cluster_map.get(p, 0)
            self.spend_ema[cid] = self.ema_alpha * abs(r) + (1 - self.ema_alpha) * self.spend_ema[cid]
            self.visit_counts[cid] += 1
        self.step += 1
        if self.step % 20 == 0:
            print(f"[CB-GRPO] step={self.step} spend_ema min/max="
                  f"{self.spend_ema.min():.3f}/{self.spend_ema.max():.3f} "
                  f"visits min/max={self.visit_counts.min()}/{self.visit_counts.max()} "
                  f"argmax_cluster={int(self.spend_ema.argmax())}")
        return rewards

class CBGRPOSampler(torch.utils.data.Sampler):
    """Lazy generator: recomputes sampling probs from LIVE spend_ema every
    `refresh_every` GROUPS. Repeats each chosen prompt `num_generations` times
    consecutively (required by GRPOTrainer's grouping). Clusters below
    `min_visits` are never gated — avoids acting on noisy early EMA estimates,
    and avoids punishing clusters just for having zero visits so far."""
    def __init__(self, cluster_ids, spend_ema, visit_counts, num_generations,
                 theta=1.2, decay=0.9, min_visits=10, refresh_every=2,
                 seed=42, log_deque=None):
        self.cluster_ids = np.array(cluster_ids)
        self.spend_ema = spend_ema
        self.visit_counts = visit_counts
        self.num_generations = num_generations
        self.theta, self.decay = theta, decay
        self.min_visits = min_visits
        self.refresh_every = refresh_every
        self.rng = np.random.default_rng(seed)
        self.n = len(cluster_ids)
        self.log_deque = log_deque

    def _recompute_probs(self):
        mean_spend = self.spend_ema.mean() + 1e-8
        ratios = self.spend_ema[self.cluster_ids] / mean_spend
        eligible = self.visit_counts[self.cluster_ids] >= self.min_visits
        w = np.where(eligible & (ratios > self.theta),
                     self.decay ** (ratios - self.theta),
                     1.0)
        return w / w.sum()

    def __iter__(self):
        probs = self._recompute_probs()
        num_groups = self.n // self.num_generations
        for g in range(num_groups):
            if g > 0 and g % self.refresh_every == 0:
                probs = self._recompute_probs()
            base_idx = int(self.rng.choice(self.n, p=probs))
            if self.log_deque is not None:
                self.log_deque.append(int(self.cluster_ids[base_idx]))
            for _ in range(self.num_generations):
                yield base_idx

    def __len__(self):
        return (self.n // self.num_generations) * self.num_generations

cb_reward = CBGRPOSpendTracker(compute_reward_for_grpo, cluster_assignments,
                                shared_spend_ema, shared_visit_counts, ema_alpha=EMA_ALPHA)

print("⏳ STEP 8/9: Initializing trainer...")

from transformers import GenerationConfig

generation_config = GenerationConfig(

    pad_token_id=tokenizer.pad_token_id, eos_token_id=tokenizer.eos_token_id,
)

class CBGRPOTrainer(GRPOTrainer):
    def __init__(self, cluster_ids_list, spend_ema, visit_counts, theta, decay,
                 min_visits, refresh_every, num_generations, sampled_log, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.cluster_ids_list = cluster_ids_list
        self.spend_ema = spend_ema
        self.visit_counts = visit_counts
        self.theta, self.decay, self.min_visits = theta, decay, min_visits
        self.refresh_every = refresh_every
        self.num_generations = num_generations
        self.sampled_log = sampled_log

    def get_train_dataloader(self):
        sampler = CBGRPOSampler(self.cluster_ids_list, self.spend_ema, self.visit_counts,
                                 num_generations=self.num_generations,
                                 theta=self.theta, decay=self.decay,
                                 min_visits=self.min_visits,
                                 refresh_every=self.refresh_every,
                                 log_deque=self.sampled_log)
        return torch.utils.data.DataLoader(
            self.train_dataset, batch_size=self._train_batch_size,
            sampler=sampler, collate_fn=self.data_collator,
            num_workers=0, pin_memory=self.args.dataloader_pin_memory,
        )

    report_to="none", seed=42, dataloader_num_workers=0,
    max_grad_norm=1.0,          # cheap insurance against the loss-spike behavior seen last run
)
