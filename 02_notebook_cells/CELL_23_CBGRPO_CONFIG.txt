# These are the key hyperparameters for CB-GRPO
# You can tune these to optimize balance

N_CLUSTERS = 16              # Number of prompt clusters (should match Cell 6)
EMA_ALPHA = 0.01             # EMA decay rate for spend tracking
                             # Lower = longer memory (0.01-0.05 recommended)

DECAY_FACTOR = 0.9           # Decay for over-spending clusters
                             # Must be < 1.0 (0.85-0.95 recommended)

THETA_THRESHOLD = 1.2        # Threshold multiplier for population mean
                             # Clusters with spend > theta * mean are penalized
                             # (1.1-1.5 recommended)

BALANCE_WINDOW = 50          # Steps between balance logging
TRACK_PERPLEXITY = True      # Track perplexity per cluster (adds overhead)

print(f"  n_clusters:        {N_CLUSTERS}")
print(f"  ema_alpha:         {EMA_ALPHA} (memory: {'long' if EMA_ALPHA < 0.02 else 'medium'})")
print(f"  decay_factor:      {DECAY_FACTOR} (penalty: {'strong' if DECAY_FACTOR < 0.9 else 'moderate'})")
print(f"  theta_threshold:   {THETA_THRESHOLD} (sensitivity: {'high' if THETA_THRESHOLD < 1.2 else 'medium'})")
print(f"  balance_window:    {BALANCE_WINDOW} steps")
print(f"  track_perplexity:  {TRACK_PERPLEXITY}")
print("─" * 80)
