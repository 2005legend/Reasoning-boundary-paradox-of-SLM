def compute_reward_for_grpo(completions, ground_truth, **kwargs):
    """
    Reward function for GRPO training.
    
    Combines format reward (XML structure) and correctness reward (answer matching).
    
    Args:
        completions: List of model generated texts
        ground_truth: List of correct answers
    
    Returns:
        List of reward scores between 0.0 and 1.0
    """
    rewards = []
    for completion, gt in zip(completions, ground_truth):
        reward = compute_total_reward(completion, gt, format_weight=0.2)
        rewards.append(reward)
    return rewards
