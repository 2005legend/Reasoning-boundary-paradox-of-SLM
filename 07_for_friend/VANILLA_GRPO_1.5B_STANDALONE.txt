# STEP 5: SETUP REWARD FUNCTION

print("⏳ STEP 5/7: Configuring reward function...")

import re
import xml.etree.ElementTree as ET

def parse_xml_robust(text, tag):
    """Robust XML parsing with fallback (same as CELL_21)"""
    # Try proper XML parsing first
    try:
        wrapped = f"<root>{text}</root>"
        root = ET.fromstring(wrapped)
        elem = root.find(tag)
        if elem is not None and elem.text:
            return elem.text.strip()
    except:
        pass
    
    # Fallback to regex
    pattern = f"<{tag}>(.*?)</{tag}>"
    match = re.search(pattern, text, re.DOTALL | re.IGNORECASE)
    if match:
        return match.group(1).strip()
    
    return None

def compute_format_reward(completion):
    """Check if completion has proper XML format"""
    has_reasoning = '<reasoning>' in completion.lower() and '</reasoning>' in completion.lower()
    has_answer = '<answer>' in completion.lower() and '</answer>' in completion.lower()
    return 1.0 if (has_reasoning and has_answer) else 0.0

def compute_correctness_reward(completion, ground_truth):
    """Check if extracted answer matches ground truth"""
    answer_text = parse_xml_robust(completion, 'answer')
    if answer_text is None:
        return 0.0
    
    # Extract number from answer
    numbers = re.findall(r'-?\d+(?:\.\d+)?', answer_text.replace(',', ''))
    if not numbers:
        return 0.0
    
    predicted = numbers[-1]  # Take last number
    
    # Normalize and compare
    try:
        pred_val = float(predicted)
        gt_val = float(ground_truth)
        return 1.0 if abs(pred_val - gt_val) < 1e-6 else 0.0
    except:
        return 0.0

def compute_reward_for_grpo(completions, ground_truth, **kwargs):
    """
    Reward function for GRPO training (same as CELL_21).
    Format weight: 0.2, Correctness weight: 0.8
    """
    rewards = []
    for completion, gt in zip(completions, ground_truth):
        format_reward = compute_format_reward(completion)
        correctness_reward = compute_correctness_reward(completion, gt)
        total_reward = 0.2 * format_reward + 0.8 * correctness_reward
        rewards.append(total_reward)
    return rewards
